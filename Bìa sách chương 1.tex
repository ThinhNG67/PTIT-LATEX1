\newcommand{\chip}[4]{% #1 tên, #2 màu, #3 vị trí, #4 chữ
  \node[chip, fill=#2!10, draw=#2!45, text=#2] (#1) at #3 {#4};}
\newcommand{\chipnext}[4]{% #1 tên, #2 màu, #3 chip trước, #4 chữ
  \node[chip, fill=#2!10, draw=#2!45, text=#2] (#1) at ($(#3.east)+(2.6,0)$) {#4};}
